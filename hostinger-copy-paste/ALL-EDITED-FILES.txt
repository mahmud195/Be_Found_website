function befound_theme_asset_url( $path ) {
	if ( empty( $path ) ) {
		return '';
	}

	if ( is_numeric( $path ) ) {
		$source_asset = get_post_meta( (int) $path, '_befound_source_asset', true );
		if ( is_string( $source_asset ) && '' !== $source_asset ) {
			return get_theme_file_uri( 'assets/' . ltrim( $source_asset, '/' ) );
		}

		$url = wp_get_attachment_url( (int) $path );
		return $url ? $url : '';
	}

	if ( is_array( $path ) && ! empty( $path['url'] ) ) {
		return $path['url'];
	}

	$path = (string) $path;

	if ( preg_match( '~^(https?:)?//|^data:|^/~', $path ) ) {
		return $path;
	}

	return get_theme_file_uri( 'assets/' . ltrim( $path, '/' ) );
}

function befound_lines_from_value( $value, $fallback ) {
	if ( is_array( $value ) ) {
		$value = implode( "\n", $value );
	}

	if ( ! is_string( $value ) || '' === trim( $value ) ) {
		return $fallback;
	}

	$lines = preg_split( '/\R+/u', trim( $value ) );
	$lines = array_values( array_filter( array_map( 'trim', $lines ) ) );

	return $lines ? $lines : $fallback;
}

function befound_setting_value( $settings, $key, $fallback = '' ) {
	if ( isset( $settings[ $key ] ) && '' !== $settings[ $key ] && array() !== $settings[ $key ] ) {
		return $settings[ $key ];
	}

	return $fallback;
}

function befound_gallery_from_value( $value, $fallback = array() ) {
	if ( is_array( $value ) ) {
		return $value;
	}

	if ( ! is_string( $value ) || '' === trim( $value ) ) {
		return $fallback;
	}

	return array_values( array_filter( array_map( 'trim', explode( ',', $value ) ) ) );
}

function befound_google_maps_search_url( $query ) {
	return 'https://www.google.com/maps/search/?api=1&query=' . rawurlencode( trim( (string) $query ) );
}

function befound_phone_digits( $phone ) {
	return preg_replace( '/\D+/', '', (string) $phone );
}

function befound_tel_url( $phone ) {
	$phone = preg_replace( '/[^\d+]/', '', (string) $phone );
	return $phone ? 'tel:' . $phone : '';
}

function befound_whatsapp_url( $phone, $message = '' ) {
	$digits = befound_phone_digits( $phone );
	if ( ! $digits ) {
		return '';
	}

	$url = 'https://wa.me/' . $digits;
	if ( '' !== trim( (string) $message ) ) {
		$url .= '?text=' . rawurlencode( trim( (string) $message ) );
	}

	return $url;
}

function befound_item_lookup_key( $value ) {
	return sanitize_title( wp_specialchars_decode( (string) $value ) );
}

function befound_default_item_for_title( $section, $title, $title_key = 'title' ) {
	static $lookups = array();

	$lookup_key = $section . ':' . $title_key;
	if ( ! isset( $lookups[ $lookup_key ] ) ) {
		$content = befound_default_content();
		$items   = $content[ $section ]['items'] ?? array();
		$lookups[ $lookup_key ] = array();

		foreach ( $items as $item ) {
			$key = befound_item_lookup_key( $item[ $title_key ] ?? '' );
			if ( '' !== $key ) {
				$lookups[ $lookup_key ][ $key ] = $item;
			}
		}
	}

	$key = befound_item_lookup_key( $title );
	return $lookups[ $lookup_key ][ $key ] ?? array();
}

function befound_filled_value( $value, $fallback = '' ) {
	return ( isset( $value ) && '' !== $value && array() !== $value ) ? $value : $fallback;
}

function befound_posts_as_services() {
	$posts = get_posts(
		array(
			'post_type'      => 'befound_service',
			'post_status'    => 'publish',
			'posts_per_page' => -1,
			'orderby'        => array(
				'menu_order' => 'ASC',
				'date'       => 'ASC',
			),
		)
	);

	if ( empty( $posts ) ) {
		return array();
	}

	return array_map(
		function ( $post ) {
			$fallback = befound_default_item_for_title( 'services', get_the_title( $post ) );
			$image = get_post_thumbnail_id( $post ) ?: get_post_meta( $post->ID, '_befound_image', true );
			return array(
				'title'       => get_the_title( $post ),
				'description' => befound_filled_value( get_post_meta( $post->ID, '_befound_description', true ), $fallback['description'] ?? '' ),
				'image'       => befound_filled_value( $image, $fallback['image'] ?? '' ),
				'accent'      => befound_filled_value( get_post_meta( $post->ID, '_befound_accent', true ), $fallback['accent'] ?? '#E6F0F0' ),
				'overlay_rgb' => befound_filled_value( get_post_meta( $post->ID, '_befound_overlay_rgb', true ), $fallback['overlay_rgb'] ?? '200, 210, 210' ),
			);
		},
		$posts
	);
}

function befound_site_content() {
	static $content = null;

	if ( null !== $content ) {
		return $content;
	}

	$content = befound_default_content();
	$settings = function_exists( 'befound_get_settings' ) ? befound_get_settings() : array();

	$content['meta']['title']       = befound_setting_value( $settings, 'meta_title', $content['meta']['title'] );
	$content['meta']['description'] = befound_setting_value( $settings, 'meta_description', $content['meta']['description'] );
	$content['meta']['og_image']    = befound_setting_value( $settings, 'og_image', $content['meta']['og_image'] );

	foreach ( array_keys( $content['tracking'] ) as $tracking_key ) {
		$content['tracking'][ $tracking_key ] = befound_setting_value( $settings, $tracking_key, $content['tracking'][ $tracking_key ] );
	}

	$content['language']['enable_switcher'] = ! empty( $settings['enable_language_switcher'] ) && '1' === (string) $settings['enable_language_switcher'];
	$content['language']['default']         = befound_setting_value( $settings, 'default_language', $content['language']['default'] );

	$content['hero']['tagline'] = befound_setting_value( $settings, 'hero_tagline', $content['hero']['tagline'] );
	$content['hero']['video']   = befound_setting_value( $settings, 'hero_video', $content['hero']['video'] );

	$content['about']['title']            = befound_setting_value( $settings, 'about_title', $content['about']['title'] );
	$content['about']['background_image'] = befound_setting_value( $settings, 'about_background_image', $content['about']['background_image'] );
	$content['about']['pattern_image']    = befound_setting_value( $settings, 'about_pattern_image', $content['about']['pattern_image'] );
	$content['about']['left_paragraphs']  = befound_lines_from_value( befound_setting_value( $settings, 'about_left_paragraphs', '' ), $content['about']['left_paragraphs'] );

function befound_admin_enqueue( $hook ) {
	$screen = get_current_screen();
	if ( ! $screen ) {
		return;
	}

	$is_befound_screen = 'toplevel_page_befound-content' === $hook || in_array( $screen->post_type, array( 'befound_service', 'befound_project', 'befound_partner' ), true );
	if ( ! $is_befound_screen ) {
		return;
	}

	wp_enqueue_media();
	wp_enqueue_script(
		'befound-admin',
		get_theme_file_uri( 'assets/js/admin.js' ),
		array( 'jquery' ),
		befound_asset_version( 'assets/js/admin.js' ),
		true
	);
	wp_enqueue_style(
		'befound-admin',
		get_theme_file_uri( 'assets/css/admin.css' ),
		array(),
		befound_asset_version( 'assets/css/admin.css' )
	);
}

function befound_sanitize_settings( $input ) {
	$current = befound_get_settings();
	$input   = is_array( $input ) ? $input : array();

	$text_fields = array(
		'meta_title',
		'gtm_id',
		'ga_id',
		'google_site_verification',
		'bing_site_verification',
		'clarity_id',
		'hero_tagline',
		'about_title',
		'services_title',
		'services_intro',
		'projects_title',
		'projects_intro',
		'partners_title',
		'contact_title',
		'contact_email_label',
		'contact_phone_label',
		'contact_whatsapp_number',
		'contact_studio_label',
		'contact_jeddah_label',
		'contact_jeddah_address',
		'contact_cairo_label',
		'contact_cairo_address',
		'footer_creative_label',
	);

	foreach ( $text_fields as $field ) {
		$current[ $field ] = sanitize_text_field( $input[ $field ] ?? '' );
	}
	$current['gtm_id'] = befound_sanitize_gtm_id( $input['gtm_id'] ?? '' );

	$textarea_fields = array(
		'meta_description',
		'about_left_paragraphs',
		'about_right_paragraphs',
		'contact_intro',
		'contact_phone_lines',
		'contact_studio_lines',
	);

	foreach ( $textarea_fields as $field ) {
		$current[ $field ] = sanitize_textarea_field( $input[ $field ] ?? '' );
	}

	$url_fields = array( 'contact_jeddah_map_url', 'contact_cairo_map_url', 'footer_aoc_url', 'footer_linkedin_url', 'footer_instagram_url', 'footer_facebook_url', 'footer_whatsapp_url' );
	foreach ( $url_fields as $field ) {
		$current[ $field ] = esc_url_raw( $input[ $field ] ?? '' );
	}

	$current['contact_email']            = sanitize_email( $input['contact_email'] ?? '' );
	$current['enable_language_switcher'] = ! empty( $input['enable_language_switcher'] ) ? '1' : '0';
	$current['default_language']         = in_array( $input['default_language'] ?? 'en', array( 'en', 'ar' ), true ) ? $input['default_language'] : 'en';

	$media_fields = array(
		'og_image',
		'hero_video',
		'about_background_image',
		'about_pattern_image',
		'contact_image',
		'footer_logo_black',
		'footer_logo_arabic_black',
		'footer_aoc_icon',
		'footer_aoc_logo',
	);

	foreach ( $media_fields as $field ) {
		$current[ $field ] = befound_sanitize_media_value( $input[ $field ] ?? '' );
	}

	return $current;
}

function befound_sanitize_media_value( $value ) {
	if ( is_numeric( $value ) ) {
		return (string) absint( $value );
	}

	return sanitize_text_field( (string) $value );
}

function befound_sanitize_gtm_id( $value ) {
	$value = strtoupper( sanitize_text_field( (string) $value ) );
	return preg_match( '/^GTM-[A-Z0-9]+$/', $value ) ? $value : '';
}

function befound_apply_contact_details_update_once() {
	$update_version = '20260907_whatsapp_contact_actions';
	if ( $update_version === get_option( 'befound_contact_details_update' ) ) {
		return;
	}

	$defaults = befound_default_content();
	$contact  = $defaults['contact'];
	$settings = get_option( 'befound_settings', array() );
	$settings = is_array( $settings ) ? $settings : array();

	if ( empty( $settings['contact_phone_label'] ) ) {
		$settings['contact_phone_label'] = $contact['phone_label'];
	}

	if ( empty( $settings['contact_phone_lines'] ) ) {
		$settings['contact_phone_lines'] = implode( "\n", $contact['phone_lines'] );
	}

	if ( empty( $settings['contact_whatsapp_number'] ) ) {
		$settings['contact_whatsapp_number'] = $contact['whatsapp_number'];
	}

	if ( empty( $settings['footer_whatsapp_url'] ) ) {
		$settings['footer_whatsapp_url'] = befound_whatsapp_url( $contact['whatsapp_number'] );
	}

	$settings['contact_jeddah_label']   = empty( $settings['contact_jeddah_label'] ) ? $contact['locations'][0]['label'] : $settings['contact_jeddah_label'];
	$settings['contact_jeddah_address'] = empty( $settings['contact_jeddah_address'] ) ? $contact['locations'][0]['address'] : $settings['contact_jeddah_address'];
	$settings['contact_jeddah_map_url'] = empty( $settings['contact_jeddah_map_url'] ) ? $contact['locations'][0]['map_url'] : $settings['contact_jeddah_map_url'];
	$settings['contact_cairo_label']    = empty( $settings['contact_cairo_label'] ) ? $contact['locations'][1]['label'] : $settings['contact_cairo_label'];
	$settings['contact_cairo_address']  = empty( $settings['contact_cairo_address'] ) ? $contact['locations'][1]['address'] : $settings['contact_cairo_address'];
	$settings['contact_cairo_map_url']  = empty( $settings['contact_cairo_map_url'] ) ? $contact['locations'][1]['map_url'] : $settings['contact_cairo_map_url'];

	if ( empty( $settings['contact_studio_label'] ) || 'Studio' === $settings['contact_studio_label'] ) {
		$settings['contact_studio_label'] = $contact['studio_label'];
	}

	$old_studio_lines = "19 Street 207 - Degla Maadi\nCairo, Egypt";
	$stored_studio_lines = str_replace( "\r\n", "\n", (string) ( $settings['contact_studio_lines'] ?? '' ) );
	if ( empty( $settings['contact_studio_lines'] ) || $old_studio_lines === $stored_studio_lines || false !== strpos( $stored_studio_lines, 'مليباري' ) ) {
		$settings['contact_studio_lines'] = implode( "\n", $contact['studio_lines'] );
	}

	update_option( 'befound_settings', wp_parse_args( $settings, befound_get_settings() ) );
	update_option( 'befound_contact_details_update', $update_version );
}
add_action( 'init', 'befound_apply_contact_details_update_once', 36 );

function befound_save_post_meta( $post_id ) {
	if ( ! isset( $_POST['befound_post_meta_nonce'] ) || ! wp_verify_nonce( sanitize_text_field( wp_unslash( $_POST['befound_post_meta_nonce'] ) ), 'befound_save_post_meta' ) ) {
		return;
	}

	if ( defined( 'DOING_AUTOSAVE' ) && DOING_AUTOSAVE ) {
		return;
	}

	if ( ! current_user_can( 'edit_post', $post_id ) ) {
		return;
	}

		);

	}

function befound_imported_default_asset_path( $value ) {
	if ( ! is_numeric( $value ) ) {
		return '';
	}

	$source_asset = get_post_meta( (int) $value, '_befound_source_asset', true );
	if ( ! is_string( $source_asset ) || '' === $source_asset ) {
		return '';
	}

	$source_asset = ltrim( $source_asset, '/' );
	$source_file  = get_theme_file_path( 'assets/' . $source_asset );

	return file_exists( $source_file ) ? $source_asset : '';
}

function befound_normalize_imported_media_value( $value ) {
	$source_asset = befound_imported_default_asset_path( $value );
	return $source_asset ? $source_asset : $value;
}

function befound_normalize_imported_gallery_value( $value ) {
	$items = array_filter( array_map( 'trim', explode( ',', (string) $value ) ) );
	if ( empty( $items ) ) {
		return $value;
	}

	$changed = false;
	$items   = array_map(
		function ( $item ) use ( &$changed ) {
			$normalized = befound_normalize_imported_media_value( $item );
			if ( $normalized !== $item ) {
				$changed = true;
			}

			return $normalized;
		},
		$items
	);

	return $changed ? implode( ',', $items ) : $value;
}

function befound_normalize_imported_default_media_once() {
	$migration_version = '20260602_theme_asset_paths';
	if ( $migration_version === get_option( 'befound_media_reference_migration' ) ) {
		return;

			$settings[ $field ] = $normalized;
			$settings_changed   = true;
		}
	}

	if ( $settings_changed ) {
		update_option( 'befound_settings', $settings );
	}

	$post_types = array( 'befound_service', 'befound_project', 'befound_partner' );
	foreach ( $post_types as $post_type ) {

			$image      = get_post_meta( $post->ID, '_befound_image', true );
			$normalized = befound_normalize_imported_media_value( $image );
			if ( $normalized !== $image ) {
				update_post_meta( $post->ID, '_befound_image', $normalized );
			}

			if ( 'befound_project' === $post_type ) {
				$gallery            = get_post_meta( $post->ID, '_befound_gallery', true );
				$normalized_gallery = befound_normalize_imported_gallery_value( $gallery );
				if ( $normalized_gallery !== $gallery ) {
					update_post_meta( $post->ID, '_befound_gallery', $normalized_gallery );
				}
			}

			$thumbnail_id = get_post_thumbnail_id( $post );
			if ( $thumbnail_id && befound_imported_default_asset_path( $thumbnail_id ) ) {
				delete_post_thumbnail( $post->ID );

	}

function befound_import_theme_asset_to_media( $asset_path ) {
	if ( empty( $asset_path ) || is_numeric( $asset_path ) ) {
		return is_numeric( $asset_path ) ? (int) $asset_path : 0;
	}

	$asset_path = ltrim( (string) $asset_path, '/' );
	if ( 0 !== strpos( $asset_path, 'images/' ) && 0 !== strpos( $asset_path, 'logos/' ) && 0 !== strpos( $asset_path, 'videos/' ) ) {
		return 0;
	}

	$existing = get_posts(
		array(
			'post_type'      => 'attachment',
			'post_status'    => 'inherit',
			'posts_per_page' => 1,
			'fields'         => 'ids',
			'meta_key'       => '_befound_source_asset',
			'meta_value'     => $asset_path,
		)
	);

	if ( $existing ) {
		return (int) $existing[0];
	}

	$source = get_theme_file_path( 'assets/' . $asset_path );
	if ( ! file_exists( $source ) || ! is_readable( $source ) ) {
		return 0;
	}

	$upload_dir = wp_upload_dir();
	if ( ! empty( $upload_dir['error'] ) ) {
		return 0;
	}

	$filename        = wp_unique_filename( $upload_dir['path'], basename( $source ) );
	$destination     = trailingslashit( $upload_dir['path'] ) . $filename;
	$destination_url = trailingslashit( $upload_dir['url'] ) . $filename;

	if ( ! copy( $source, $destination ) ) {
		return 0;
	}

	$filetype = wp_check_filetype( $filename, null );
	$attachment_id = wp_insert_attachment(
		array(
			'guid'           => $destination_url,
			'post_mime_type' => $filetype['type'],
			'post_title'     => preg_replace( '/\.[^.]+$/', '', $filename ),
			'post_content'   => '',
			'post_status'    => 'inherit',
		),
		$destination
	);

	if ( is_wp_error( $attachment_id ) || ! $attachment_id ) {
		return 0;
	}

	update_post_meta( $attachment_id, '_befound_source_asset', $asset_path );

	if ( 0 === strpos( (string) $filetype['type'], 'image/' ) ) {
		require_once ABSPATH . 'wp-admin/includes/image.php';
		$metadata = wp_generate_attachment_metadata( $attachment_id, $destination );
		wp_update_attachment_metadata( $attachment_id, $metadata );
	}

	return (int) $attachment_id;
}

function befound_maybe_upgrade_home_page_to_blocks( $page_id ) {
	if ( ! function_exists( 'befound_home_blocks_content' ) || ! function_exists( 'befound_is_legacy_home_content' ) ) {
		return;
	}

	$post_content = (string) get_post_field( 'post_content', $page_id );
	if ( befound_is_befound_blocks_page( $post_content ) ) {
		return;
	}

	if ( '' !== trim( $post_content ) && ! befound_is_legacy_home_content( $post_content ) ) {
		return;
	}

	wp_update_post(
		array(
			'ID'           => $page_id,
			'post_content' => befound_home_blocks_content(),
		)
	);
}

        if (offset === 0) openModal(index);
      });
    });

    qsa('[data-project-open]', slider).forEach((button) => {
      button.addEventListener('click', (event) => {
        event.stopPropagation();
        if (lastDragDistance >= 5) return;
        openModal(parseInt(button.getAttribute('data-index') || '0', 10));
      });
    });

    slider.addEventListener('mouseenter', () => {
      isHovered = true;
    });
    slider.addEventListener('mouseleave', () => {
      isHovered = false;
    });

    slider.addEventListener('pointerdown', (event) => {
      if (event.target.closest('button')) return;
      isDragging = true;
      dragStartX = event.clientX;
      dragOffset = 0;
      lastDragDistance = 0;
    });

    slider.addEventListener('pointermove', (event) => {
      if (!isDragging) return;
      dragOffset = event.clientX - dragStartX;
      updateCards();
    });

    const endDrag = () => {
      if (!isDragging) return;
      isDragging = false;
      lastDragDistance = Math.abs(dragOffset);
      if (dragOffset > 50) goToSlide(currentIndex - 1);
      else if (dragOffset < -50) goToSlide(currentIndex + 1);
      dragOffset = 0;
      updateCards();
      window.setTimeout(() => {
        lastDragDistance = 0;
      }, 80);
    };

    slider.addEventListener('pointerup', endDrag);
    slider.addEventListener('pointerleave', endDrag);
    slider.addEventListener('pointercancel', endDrag);

    window.setInterval(() => {
      if (isHovered || isDragging || isModalOpen || document.hidden) return;
      goToSlide(currentIndex + 1);
    }, 5000);

    updateCards();
  };

  const updateSwipeDots = (dots, activeIndex) => {
    dots.forEach((dot, index) => {
      const inner = qs('[data-swipe-dot-inner]', dot);
      const active = qs('[data-swipe-dot-active]', dot);
      classSwap(inner, index === activeIndex ? ['bg-white'] : ['bg-white/30', 'hover:bg-white/60'], index === activeIndex ? ['bg-white/30', 'hover:bg-white/60'] : ['bg-white']);
      classSwap(active, index === activeIndex ? [] : ['hidden'], index === activeIndex ? ['hidden'] : []);
    });
  };

  const initSwipeGallery = (gallery) => {
    const imagesScript = qs('[data-gallery-images]', gallery);
    const stage = qs('[data-swipe-stage]', gallery);
    const currentImage = qs('[data-swipe-current]', gallery);
    const pendingImage = qs('[data-swipe-pending]', gallery);
    const dots = qsa('[data-swipe-dot]', gallery);
    const alt = gallery.getAttribute('data-gallery-alt') || '';
    if (!imagesScript || !stage || !currentImage || !pendingImage) return;

    let images = [];
    try {
      images = JSON.parse(imagesScript.textContent || '[]');
    } catch (error) {
      images = [];
    }
    if (!images.length) return;

    let currentIndex = 0;
    let displayIndex = 0;
    let isDragging = false;
    let isAnimating = false;
    let dragX = 0;
    let startX = 0;

    const setCurrentImage = (index) => {
      currentImage.src = images[index];
      currentImage.alt = `${alt} ${index + 1}`;
      currentImage.style.transform = 'translateX(0)';
      currentImage.style.opacity = '1';
      currentImage.style.transition = 'none';
      pendingImage.classList.add('hidden');
      displayIndex = index;
      currentIndex = index;
      updateSwipeDots(dots, index);
      requestAnimationFrame(() => {
        currentImage.style.transition = 'transform 0.4s ease-out, opacity 0.4s ease-out';
      });
    };

    const triggerTransition = (toIndex, direction) => {
      if (isAnimating || toIndex === currentIndex || images.length <= 1) return;
      isAnimating = true;
      pendingImage.src = images[toIndex];
      pendingImage.alt = `${alt} ${toIndex + 1}`;
      pendingImage.classList.remove('hidden');
      pendingImage.style.zIndex = '1';
      currentImage.style.zIndex = '2';
      currentImage.style.transition = 'transform 0.4s ease-out, opacity 0.4s ease-out';
      currentImage.style.transform = direction === 'next' ? 'translateX(-50%)' : 'translateX(50%)';
      currentImage.style.opacity = '0';
      updateSwipeDots(dots, toIndex);
      window.setTimeout(() => {
        setCurrentImage(toIndex);
        isAnimating = false;
      }, 400);
    };

    const prev = () => triggerTransition(currentIndex > 0 ? currentIndex - 1 : images.length - 1, 'prev');
    const next = () => triggerTransition(currentIndex < images.length - 1 ? currentIndex + 1 : 0, 'next');

    const finishDrag = () => {
      if (!isDragging) return;
      isDragging = false;
      if (Math.abs(dragX) < 50 || isAnimating) {
        currentImage.style.transform = 'translateX(0)';
        currentImage.style.opacity = '1';
        dragX = 0;
        return;
      }
      if (dragX < 0) next();
      else prev();
      dragX = 0;
    };

    stage.addEventListener('mousedown', (event) => {
      if (isAnimating || images.length <= 1) return;
      isDragging = true;
      startX = event.clientX;
      dragX = 0;
      currentImage.style.transition = 'none';
    });
    stage.addEventListener('mousemove', (event) => {
      if (!isDragging) return;
      dragX = event.clientX - startX;
      currentImage.style.transform = `translateX(${dragX}px)`;
      currentImage.style.opacity = String(Math.max(0.3, 1 - Math.abs(dragX) / 300));
    });
    stage.addEventListener('mouseup', finishDrag);
    stage.addEventListener('mouseleave', finishDrag);
    stage.addEventListener('touchstart', (event) => {
      if (isAnimating || images.length <= 1) return;
      isDragging = true;
      startX = event.touches[0].clientX;
      dragX = 0;
      currentImage.style.transition = 'none';
    }, { passive: true });
    stage.addEventListener('touchmove', (event) => {
      if (!isDragging) return;
      dragX = event.touches[0].clientX - startX;
      currentImage.style.transform = `translateX(${dragX}px)`;
      currentImage.style.opacity = String(Math.max(0.3, 1 - Math.abs(dragX) / 300));
    }, { passive: true });
    stage.addEventListener('touchend', finishDrag);

    qs('[data-swipe-prev]', gallery)?.addEventListener('click', (event) => {
      event.stopPropagation();
      prev();
    });
    qs('[data-swipe-next]', gallery)?.addEventListener('click', (event) => {
      event.stopPropagation();
      next();
    });
    qs('[data-swipe-fullscreen]', gallery)?.addEventListener('click', (event) => {
      event.stopPropagation();
      openLightbox(images, currentIndex, alt);
    });

    dots.forEach((dot) => {
      dot.addEventListener('click', () => {
        const toIndex = parseInt(dot.getAttribute('data-index') || '0', 10);
        triggerTransition(toIndex, toIndex > displayIndex ? 'next' : 'prev');
      });
    });

    setCurrentImage(0);
  };

  const openLightbox = (images, initialIndex, alt) => {
    let currentIndex = initialIndex;
    let zoom = 1;
    let pan = { x: 0, y: 0 };
    let isPanning = false;
    let panStart = { x: 0, y: 0 };
    let panOffset = { x: 0, y: 0 };

    const overlay = document.createElement('div');
    overlay.className = 'fixed inset-0 z-[200] bg-black/95 flex items-center justify-center';
    overlay.innerHTML = `
      <button type="button" class="absolute top-4 right-4 z-[210] w-10 h-10 rounded-full border border-white/30 flex items-center justify-center text-white hover:border-white hover:bg-white/10 transition-colors" data-lightbox-close>${iconSvg('x', 'w-5 h-5')}</button>
      <div class="absolute top-4 left-1/2 -translate-x-1/2 z-[210] flex items-center gap-3">
        <button type="button" class="w-9 h-9 rounded-full border border-white/30 flex items-center justify-center text-white hover:bg-white/10 transition-colors disabled:opacity-30 disabled:pointer-events-none" data-lightbox-zoom-out>${iconSvg('zoomOut', 'w-4 h-4')}</button>
        <span class="text-white/70 text-sm min-w-[3rem] text-center" style="font-family: 'Gambetta', serif" data-lightbox-zoom-label>100%</span>
        <button type="button" class="w-9 h-9 rounded-full border border-white/30 flex items-center justify-center text-white hover:bg-white/10 transition-colors disabled:opacity-30 disabled:pointer-events-none" data-lightbox-zoom-in>${iconSvg('zoomIn', 'w-4 h-4')}</button>
      </div>
      <button type="button" class="absolute left-4 md:left-8 top-1/2 -translate-y-1/2 z-[210] w-10 h-10 md:w-12 md:h-12 rounded-full border border-white/30 flex items-center justify-center text-white hover:bg-white/10 transition-colors bg-black/40 ${images.length > 1 ? '' : 'hidden'}" data-lightbox-prev>${iconSvg('chevronLeft', 'w-5 h-5')}</button>
      <button type="button" class="absolute right-4 md:right-8 top-1/2 -translate-y-1/2 z-[210] w-10 h-10 md:w-12 md:h-12 rounded-full border border-white/30 flex items-center justify-center text-white hover:bg-white/10 transition-colors bg-black/40 ${images.length > 1 ? '' : 'hidden'}" data-lightbox-next>${iconSvg('chevronRight', 'w-5 h-5')}</button>
      <div class="w-full h-full flex items-center justify-center overflow-hidden" data-lightbox-pan>
        <img draggable="false" class="max-w-[90vw] max-h-[85vh] object-contain select-none pointer-events-none" data-lightbox-image>
      </div>
      <div class="absolute bottom-6 left-1/2 -translate-x-1/2 text-white/40 text-xs tracking-widest pointer-events-none" style="font-family: 'Gambetta', serif" data-lightbox-help>SCROLL OR USE +/- TO ZOOM</div>
    `;

    document.body.appendChild(overlay);
    const image = qs('[data-lightbox-image]', overlay);
    const zoomLabel = qs('[data-lightbox-zoom-label]', overlay);
    const zoomOutButton = qs('[data-lightbox-zoom-out]', overlay);
    const zoomInButton = qs('[data-lightbox-zoom-in]', overlay);
    const panSurface = qs('[data-lightbox-pan]', overlay);
    const help = qs('[data-lightbox-help]', overlay);

    const update = () => {
      image.src = images[currentIndex];
      image.alt = `${alt} ${currentIndex + 1}`;
      image.style.transform = `translate(${pan.x}px, ${pan.y}px) scale(${zoom})`;
      image.style.transition = isPanning ? 'none' : 'transform 0.2s ease';
      zoomLabel.textContent = `${Math.round(zoom * 100)}%`;
      zoomOutButton.disabled = zoom <= 1;
      zoomInButton.disabled = zoom >= 5;
      panSurface.style.cursor = zoom > 1 ? (isPanning ? 'grabbing' : 'grab') : 'default';
      help.textContent = zoom <= 1 ? 'SCROLL OR USE +/- TO ZOOM' : 'DRAG TO PAN \u00b7 SCROLL TO ZOOM';
    };

    const goPrev = () => {
      currentIndex = currentIndex === 0 ? images.length - 1 : currentIndex - 1;
      zoom = 1;
      pan = { x: 0, y: 0 };
      update();
    };

    const goNext = () => {
      currentIndex = currentIndex === images.length - 1 ? 0 : currentIndex + 1;
      zoom = 1;
      pan = { x: 0, y: 0 };
      update();
    };

    const zoomIn = () => {
      zoom = Math.min(zoom + 0.5, 5);
      update();
    };

    const zoomOut = () => {
      zoom = Math.max(zoom - 0.5, 1);
      if (zoom === 1) pan = { x: 0, y: 0 };
      update();
    };

    const close = () => {
      window.removeEventListener('keydown', onKey);
      overlay.remove();
    };

    const onKey = (event) => {
      if (event.key === 'Escape') close();
      else if (event.key === 'ArrowLeft') goPrev();
      else if (event.key === 'ArrowRight') goNext();
      else if (event.key === '+' || event.key === '=') zoomIn();
      else if (event.key === '-') zoomOut();
    };

    overlay.addEventListener('wheel', (event) => {
      event.preventDefault();
      if (event.deltaY < 0) {
        zoom = Math.min(zoom + 0.25, 5);
      } else {
        zoom = Math.max(zoom - 0.25, 1);
        if (zoom === 1) pan = { x: 0, y: 0 };
      }
      update();
    }, { passive: false });

    panSurface.addEventListener('mousedown', (event) => {
      if (zoom <= 1) return;
      event.preventDefault();
      isPanning = true;
      panStart = { x: event.clientX, y: event.clientY };
      panOffset = { ...pan };
      update();
    });

    panSurface.addEventListener('mousemove', (event) => {
      if (!isPanning) return;
      pan = {
        x: panOffset.x + (event.clientX - panStart.x),
        y: panOffset.y + (event.clientY - panStart.y),
      };
      update();
    });

    const stopPan = () => {
      isPanning = false;
      update();
    };
    panSurface.addEventListener('mouseup', stopPan);
    panSurface.addEventListener('mouseleave', stopPan);

    qs('[data-lightbox-close]', overlay).addEventListener('click', close);
    qs('[data-lightbox-prev]', overlay).addEventListener('click', goPrev);
    qs('[data-lightbox-next]', overlay).addEventListener('click', goNext);
    zoomInButton.addEventListener('click', zoomIn);
    zoomOutButton.addEventListener('click', zoomOut);
    window.addEventListener('keydown', onKey);
    update();
  };

  const initContactPhoneActions = () => {
    const phoneItems = qsa('[data-contact-phone]');
    if (!phoneItems.length) return;

    const closeItem = (item) => {
      item.classList.remove('is-open');
      const trigger = qs('[data-contact-phone-trigger]', item);
      if (trigger) trigger.setAttribute('aria-expanded', 'false');
    };

    const closeAll = (except = null) => {
      phoneItems.forEach((item) => {
        if (item !== except) closeItem(item);
      });
    };

    phoneItems.forEach((item) => {
      const trigger = qs('[data-contact-phone-trigger]', item);
      if (!trigger) return;

      trigger.addEventListener('click', (event) => {
        event.preventDefault();
        const isOpen = item.classList.contains('is-open');
        closeAll(item);
        item.classList.toggle('is-open', !isOpen);
        trigger.setAttribute('aria-expanded', String(!isOpen));
      });
    });

    document.addEventListener('click', (event) => {
      if (!event.target.closest('[data-contact-phone]')) {
        closeAll();
      }
    });

    document.addEventListener('keydown', (event) => {
      if (event.key === 'Escape') closeAll();
    });
  };

  const initLanguageReadiness = () => {
    const config = window.BeFoundTheme && window.BeFoundTheme.language;
    if (!config || !config.enable_switcher) return;

    if (config.default === 'ar') {
      document.documentElement.setAttribute('lang', 'ar');
      document.documentElement.setAttribute('dir', 'rtl');
    } else {
      document.documentElement.setAttribute('lang', 'en');
      document.documentElement.setAttribute('dir', 'ltr');
    }
  };

  document.addEventListener('DOMContentLoaded', () => {
    initLanguageReadiness();
    initNavbar();
    initReveals();
    initAboutToggle();
    initServices();
    initProjects();
    initContactPhoneActions();
    qsa('[data-swipe-gallery]').forEach(initSwipeGallery);
  });
})();

function befound_asset_version( $relative_path ) {
	$file = get_theme_file_path( $relative_path );
	return file_exists( $file ) ? (string) filemtime( $file ) : '1.0.0';
}

function befound_enqueue_assets() {
	wp_enqueue_style(
		'befound-theme',
		get_stylesheet_uri(),
		array(),
		befound_asset_version( 'style.css' )
	);

	wp_enqueue_style(
		'befound-main',
		get_theme_file_uri( 'assets/css/befound.css' ),
		array( 'befound-theme' ),
		befound_asset_version( 'assets/css/befound.css' )
	);

	wp_enqueue_style(
		'befound-contact-actions',
		get_theme_file_uri( 'assets/css/contact-actions.css' ),
		array( 'befound-main' ),
		befound_asset_version( 'assets/css/contact-actions.css' )
	);

	wp_enqueue_script(
		'befound-main',
		get_theme_file_uri( 'assets/js/befound.js' ),
		array(),
		befound_asset_version( 'assets/js/befound.js' ),
		true
	);

	wp_localize_script(
		'befound-main',
		'BeFoundTheme',
		array(
			'language' => befound_site_content()['language'],
		)
	);
}
add_action( 'wp_enqueue_scripts', 'befound_enqueue_assets' );
